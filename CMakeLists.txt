cmake_minimum_required(VERSION 3.25)
project(quarry VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(QUARRY_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
include(CheckCXXSourceCompiles)
check_cxx_source_compiles("int main(){ __int128 x=1; return int(x-1); }" QUARRY_HAS_INT128)
if(NOT QUARRY_HAS_INT128)
  message(FATAL_ERROR "Quarry requires checked 128-bit integer support")
endif()
if(NOT EXISTS "${CMAKE_SOURCE_DIR}/.deps/include/json.hpp" OR NOT EXISTS "${CMAKE_SOURCE_DIR}/.deps/include/doctest.h")
  message(FATAL_ERROR "Run python3.12 tools/bootstrap.py first (ordinary builds are offline)")
endif()
add_library(quarry_core src/common/value.cpp src/storage/storage.cpp src/sql/sql.cpp src/planner/planner.cpp src/execution/scalar/executor.cpp src/execution/vector/executor.cpp)
target_include_directories(quarry_core PUBLIC include PRIVATE .deps/include src)
target_compile_options(quarry_core PRIVATE -Wall -Wextra -Wconversion -Wshadow)
add_executable(quarry src/cli/main.cpp src/cli/bench.cpp)
target_link_libraries(quarry PRIVATE quarry_core)
target_include_directories(quarry PRIVATE .deps/include)
target_compile_definitions(quarry PRIVATE QUARRY_EXAMPLE_DIR="${CMAKE_SOURCE_DIR}/examples" QUARRY_BUILD_TYPE="${CMAKE_BUILD_TYPE}")
if(QUARRY_SANITIZERS)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    message(FATAL_ERROR "Sanitizers are unsupported for this compiler")
  endif()
  target_compile_options(quarry_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(quarry_core PUBLIC -fsanitize=address,undefined)
endif()
enable_testing()
add_executable(quarry_tests tests/unit/core.cpp)
target_include_directories(quarry_tests PRIVATE .deps/include src)
target_link_libraries(quarry_tests PRIVATE quarry_core)
add_test(NAME quarry_unit COMMAND quarry_tests)

add_executable(quarry_fuzz tests/fuzz/native.cpp)
target_link_libraries(quarry_fuzz PRIVATE quarry_core)
target_compile_definitions(quarry_fuzz PRIVATE QUARRY_FUZZ_ARTIFACT_DIR="${CMAKE_BINARY_DIR}/fuzz")
add_test(NAME quarry_bounded_fuzz COMMAND quarry_fuzz)
set_tests_properties(quarry_bounded_fuzz PROPERTIES TIMEOUT 60)
include(GNUInstallDirs)
install(TARGETS quarry quarry_core RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/quarry DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(DIRECTORY examples/ DESTINATION ${CMAKE_INSTALL_DATADIR}/quarry/examples)

if(APPLE)
  set(QUARRY_TEST_ASAN_OPTIONS "detect_leaks=0")
else()
  set(QUARRY_TEST_ASAN_OPTIONS "detect_leaks=1")
endif()
set_tests_properties(quarry_unit quarry_bounded_fuzz PROPERTIES ENVIRONMENT
  "ASAN_OPTIONS=${QUARRY_TEST_ASAN_OPTIONS};UBSAN_OPTIONS=halt_on_error=1:print_stacktrace=1")
